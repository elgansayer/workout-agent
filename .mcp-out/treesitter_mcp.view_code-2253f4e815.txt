 353 |         self, state: AccountState, session: Mapping[str, Any]
 354 |     ) -> DenialReason | None:
 355 |         status = state["status"]
 356 |         if status != ACCOUNT_STATUS_ACTIVE:
 357 |             return _STATUS_REASONS.get(status, DenialReason.ACCOUNT_STATE_INVALID)
 358 |         if not self.config.allows(state["email"].strip().lower()):
 359 |             return DenialReason.NOT_ALLOWLISTED
 360 |         # Sessions issued before this feature carry no epoch; they are valid
 361 |         # exactly until the first revocation advances the stored epoch.
 362 |         issued = session.get(SESSION_EPOCH_KEY, 0)
 363 |         if isinstance(issued, bool) or not isinstance(issued, int):
 364 |             return DenialReason.SESSION_REVOKED
