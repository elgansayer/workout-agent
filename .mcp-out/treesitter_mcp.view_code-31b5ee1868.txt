 210 | def _prescription_preserved(baseline: str, candidate: str) -> bool:
 211 |     """Reject generated prose that drops or rewrites a prescribed exercise/scheme."""
 212 | 
 213 |     def parse(text: str) -> list[tuple[str, str]]:
 214 |         parsed: list[tuple[str, str]] = []
 215 |         for line in text.splitlines()[1:]:
 216 |             if ":" not in line:
 217 |                 continue
 218 |             name, scheme = line.split(":", 1)
 219 |             scheme = scheme.split("(", 1)[0].strip()
 220 |             scheme = " ".join(scheme.replace("×", "x").lower().split())
 221 |             parsed.append((name.strip().lower(), scheme))

 223 | 
 224 |     expected = parse(baseline)
 225 |     actual = parse(candidate)
