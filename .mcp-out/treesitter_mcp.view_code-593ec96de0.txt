 161 | def seal(
 162 |     master: MasterKey,
 163 |     *,
 164 |     user_id: str,
 165 |     provider: str,
 166 |     field_name: str,
 167 |     plaintext: str,
 168 | ) -> str:
 169 |     """Encrypt *plaintext* for one user's credential field.
 170 | 
 171 |     The result is ``wv2.<key id>.<base64url(nonce || ciphertext || tag)>``.
 172 |     """
 173 |     aad = _binding(user_id, provider, field_name)
 174 |     nonce = os.urandom(_NONCE_BYTES)
 175 |     sealed = AESGCM(_user_key(master, user_id)).encrypt(
 176 |         nonce,
 177 |         plaintext.encode("utf-8"),
 178 |         aad,
 179 |     )
 180 |     body = base64.urlsafe_b64encode(nonce + sealed).rstrip(b"=").decode("ascii")
 181 |     return f"{SEALED_PREFIX}.{master.key_id}.{body}"

 184 | def open_sealed(
 185 |     master: MasterKey,
 186 |     *,
 187 |     user_id: str,
 188 |     provider: str,
 189 |     field_name: str,
 190 |     token: str,
 191 | ) -> str:
 192 |     """Decrypt a value produced by :func:`seal` for the same binding.
 193 | 
 194 |     Raises :class:`DecryptionError` when the token is malformed, was sealed under
 195 |     a different master key, belongs to a different user/provider/field, or was
 196 |     tampered with.
 197 |     """
 198 |     aad = _binding(user_id, provider, field_name)
 199 |     parts = token.split(".")
 200 |     if len(parts) != 3 or parts[0] != SEALED_PREFIX:
 201 |         raise DecryptionError("malformed")
 202 |     if parts[1] != master.key_id:
 203 |         raise DecryptionError("key_mismatch")
 204 |     try:
 205 |         padded = parts[2] + "=" * (-len(parts[2]) % 4)
 206 |         blob = base64.urlsafe_b64decode(padded.encode("ascii"))
 207 |     except (binascii.Error, UnicodeEncodeError, ValueError):
 208 |         raise DecryptionError("malformed") from None
 209 |     if len(blob) < _NONCE_BYTES + _TAG_BYTES:
 210 |         raise DecryptionError("malformed")
 211 | 
 212 |     try:
 213 |         plaintext = AESGCM(_user_key(master, user_id)).decrypt(
 214 |             blob[:_NONCE_BYTES],
 215 |             blob[_NONCE_BYTES:],
 216 |             aad,
 217 |         )
 218 |         return plaintext.decode("utf-8")
 219 |     except InvalidTag:
 220 |         raise DecryptionError("authentication_failed") from None
 221 |     except UnicodeDecodeError:
 222 |         raise DecryptionError("malformed") from None
