 107 |     """Non-blocking in-process locks keyed by owner and job.
 108 | 
 109 |     A lock never spans owners: one user's slow job cannot delay or suppress

 112 |     """

 118 |     @contextmanager
 119 |     def try_acquire(self, key: str) -> Iterator[bool]:
 120 |         with self._guard:
 121 |             acquired = key not in self._held
